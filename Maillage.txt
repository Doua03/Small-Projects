void updateCamera() {
    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();

}

bool checkCollision(float x1, float y1, float z1, float radius1, float x2, float y2, float z2, float radius2) {
    float distance = sqrt(pow(x2 - x1, 2) + pow(y2 - y1, 2) + pow(z2 - z1, 2));
    return distance < (radius1 + radius2);
}

void Draw3DCube(float x, float y, float z, float a, float b, float c,float r , float g, float q) {
    glPushMatrix();
    glTranslated(x, y, z);
    glScalef(a, b, c);
    glColor3d(r, g, q);  // Green color
    glutSolidCube(1);
    glPopMatrix();
}
void Draw3DSphere(float x, float y, float z, float radius,float r,float g,float q) {
    glPushMatrix();
    glTranslated(x, y, z);
    glColor3d(r,g,q);

}

void drawText3D(float x, float y, float z, const char* text, float r, float g, float b) {
    glPushMatrix();
    glColor3f(r, g, b);
    glRasterPos3f(x, y, z);

    for (int i = 0; text[i] != '\0'; i++) {
        glutBitmapCharacter(GLUT_BITMAP_HELVETICA_18, text[i]);
    }

    glPopMatrix();
}
void display() {
    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
    updateCamera();

    glEnable(GL_DEPTH_TEST);

    if (gameState == PLAYING) {
        nuage(0.6,1.6,0,0.15);
    nuage(1.8,2.5,0,0.15);
    nuage(-1.2,2.3,0,0.15);
    nuage(-2.3,1.2,0,0.15);

    tree(0,0.6,0.02);
    tree(1.5,0.6,0.02);
    tree(-1.5,0.6,0.02);
    envi();
    obstacle(obstacleX,obstacleY,obstacleZ);
     obstacle(obstacle1X,obstacle1Y,obstacle1Z);
      obstacle(obstacle2X,obstacle2Y,obstacle2Z);
    // Move the rabbit based on arrow key input
    drawRabbit(rabbitX, rabbitY, rabbitZ);
   // Check for collision with obstacles
        if (checkCollision(rabbitX, rabbitY, rabbitZ, 0.08, obstacleX, obstacleY, obstacleZ, 0.1) ||
            checkCollision(rabbitX, rabbitY, rabbitZ, 0.08, obstacle1X, obstacle1Y, obstacle1Z, 0.1) ||
            checkCollision(rabbitX, rabbitY, rabbitZ, 0.08, obstacle2X, obstacle2Y, obstacle2Z, 0.1)) {
            gameState = GAME_OVER;
        }

        // Check for collision with finish line
        if (rabbitX < -2.6 && rabbitY > -2.7 && rabbitY < -1.7) {
            gameState = YOU_WON;
        }
    } else if (gameState == GAME_OVER) {
        // Display the game-over message
        glColor3f(0, 0, 0);  // Set color to red
        glRasterPos2f(-0.5, 0);
        std::string gameOverText = "Game Over! Click to restart.";
        for (size_t i = 0; i < gameOverText.length(); ++i) {
            glutBitmapCharacter(GLUT_BITMAP_HELVETICA_18, gameOverText[i]);
        }

        // Start a timer to control how long the game-over message is displayed
        glutTimerFunc(2000, [](int value) {
            // Reset the game and hide the game-over message after 2 seconds
            resetGame();
            gameState = PLAYING;
            glutPostRedisplay();
        }, 0);
    } else if (gameState == YOU_WON) {
        // Draw a scene for winning
        glColor3f(0, 0, 0);  // Set color to green
        glRasterPos2f(-0.5, 0);
        std::string youWonText = "Congratulations! You won! Click to restart.";
        for (size_t i = 0; i < youWonText.length(); ++i) {
            glutBitmapCharacter(GLUT_BITMAP_HELVETICA_18, youWonText[i]);
        }

        // Start a timer to control how long the "you won" message is displayed
        glutTimerFunc(5000, [](int value) {
            // Reset the game and hide the "you won" message after 5 seconds
            resetGame();
            gameState = PLAYING;
            glutPostRedisplay();
        }, 0);
    }

    glDisable(GL_DEPTH_TEST);

    glFlush();
    glutSwapBuffers();
}

void resetGame() {
    // Reset rabbit and obstacle positions, camera, and other game variables
    rabbitX = 0.0;
    rabbitY = 0.0;
    rabbitZ = 0.7;

    obstacleX = 1.2;
    obstacleY = -0.8;
    obstacleZ = 0.7;

    // Reset camera
    cameraX = 0.0;
    cameraY = 0.0;
    cameraZ = 3.0;

    // Reset other game variables
    gameState = PLAYING;
}


void keyboardCallback(unsigned char key, int x, int y) {
    switch (key) {
        case 's':
            happy = true;   // Set to smiling when 's' is pressed
            break;
        case 't':
            happy = false;  // Set to sad when 't' is pressed
            break;
        case 'i':
            // Start or stop bouncing when 'i' is pressed
            bouncing = !bouncing;
            break;
            case 'R':
        case 'r':
           changeRabbitColor(1.0, 0.0, 0.0); // Red
            break;
        case 'G':
    case 'g':
        changeRabbitColor(0.0, 1.0, 0.0); // Green
        break;
    case 'B':
    case 'b':
        changeRabbitColor(0.0, 0.0, 1.0); // Blue
        break;
         // Camera controls
        case 'U':
        case 'u':
            cameraZ += 0.1; // Move camera along positive Z
            break;
        case 'D':
        case 'd':
            cameraZ -= 0.1; // Move camera along negative Z
            break;
        case 'P':
        case 'p':
            cameraX += 0.1; // Move camera along positive X
            break;
        case 'N':
        case 'n':
            cameraX -= 0.1; // Move camera along negative X
            break;
        case 'X':
        case 'x':
            cameraY += 0.1; // Move camera along positive Y
            break;
        case 'L':
        case 'l':
            cameraY -= 0.1; // Move camera along negative Y
            break;
    }
    updateCamera(); // Update camera after any key press
    glutPostRedisplay();  // Request a redraw to reflect the changes
}

void specialKeys(int key, int x, int y) {
    switch (key) {
        case GLUT_KEY_LEFT:
            rabbitX -= 0.1;  // Move the rabbit left
            break;
        case GLUT_KEY_RIGHT:
            rabbitX += 0.1;  // Move the rabbit right
            break;
        case GLUT_KEY_UP:
            rabbitY += 0.1;  // Move the rabbit up
            break;
        case GLUT_KEY_DOWN:
            rabbitY -= 0.1;  // Move the rabbit down
            break;
        // You can handle other keys as needed
    }
    glutPostRedisplay();  // Request a redraw to reflect the changes
}

void idle() {
    if (bouncing) {
        // Bounce the rabbit using a sine function
        rabbitY = 0.4 + 0.1 * sin(glutGet(GLUT_ELAPSED_TIME) / 100.0);
    }
    glutPostRedisplay();
}
void Reshape(int width, int height) {
    if (height == 0) height = 1;
    glViewport(0, 0, width, height);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    glOrtho(-3.0, 3.0, -3.0, 3.0, 0.1, 100.0); // Adjust these values based on your scene's dimensions
    glMatrixMode(GL_MODELVIEW);
}
void mouseCallback(int button, int state, int x, int y) {
    // Handle mouse click events
    if (gameState == GAME_OVER || gameState == YOU_WON) {
        resetGame();
        glutPostRedisplay();
    }
}
int main(int argc, char** argv) {
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB | GLUT_DEPTH);
    glutInitWindowSize(800, 600);
    glutCreateWindow("Rabbit 3D");
    glEnable(GL_DEPTH_TEST);
    glClearColor(0.6f, 0.6f, 1.0f, 1.0f);  // Light blue clear color

    glutDisplayFunc(display);
     glutReshapeFunc(Reshape);
    glutKeyboardFunc(keyboardCallback);
    // Register the mouse callback function
    glutMouseFunc(mouseCallback);

    glutSpecialFunc(specialKeys); // Register the specialKeys function
       // Set up the idle function
    glutIdleFunc(idle);


    glutMainLoop();
    return 0;
}
